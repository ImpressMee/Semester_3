# Berechnung des Anstieges von 1990 - 2015
verfügbare_jahre = df['Incident Year'].unique()
print(f"Verfügbare Jahre: {sorted(verfügbare_jahre)}")

if 1990 in verfügbare_jahre and 2015 in verfügbare_jahre:
    birdstrikes_1990 = df[df['Incident Year'] == 1990].shape[0]
    birdstrikes_2015 = df[df['Incident Year'] == 2015].shape[0]
    
    anstieg_absolut = birdstrikes_2015 - birdstrikes_1990
    anstieg_prozent = ((birdstrikes_2015 - birdstrikes_1990) / birdstrikes_1990) * 100
    
    print(f"\nAnstieg 1990-2015: +{anstieg_absolut} (+{anstieg_prozent:.1f}%)")
else:
    print("Fehlende Daten für 1990 oder 2015")

# Welche Vögel verursachen den meisten Schaden?
damage_columns = [col for col in df.columns if 'Damage' in col]

# Filtere nur Vögel (keine Rehe/Säugetiere)
voegel_ohne_saeugetiere = df[~df['Species Name'].str.contains('DEER', case=False, na=False)]

# Daten Speichern
schaden_pro_vogel = voegel_ohne_saeugetiere.groupby('Species Name')[damage_columns].sum()  # Summiere Schäden pro Vogel
gesamt_schaden = schaden_pro_vogel.sum(axis=1)  # Summiere alle Schadensarten pro Vogel
top_10_voegel = gesamt_schaden.sort_values(ascending=False).head(10)  # Sortiere und nimm Top 10

# Plot erstellen
plt.figure(figsize=(10, 6))
plt.bar(top_10_voegel.index, top_10_voegel.values, color='red')
plt.title("Top 10 Vögel mit dem meisten Schäden")
plt.xlabel("Vogelarten")
plt.ylabel("Insgesamt verursachter Schaden")
plt.xticks(rotation=80)
plt.grid(axis='y', linestyle='--')
plt.tight_layout()
plt.show()
